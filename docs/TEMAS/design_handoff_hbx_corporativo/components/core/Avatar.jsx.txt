import React from "react";

/**
 * Avatar — initials chip with the HBX brand gradient, or an image.
 * Use for users, leads and companies.
 */
function Avatar({ name = "", src = null, size = 40, tone = "brand", style, ...rest }) {
  const initials = name
    .split(/\s+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((w) => w[0])
    .join("")
    .toUpperCase();

  const gradient =
    tone === "accent"
      ? "linear-gradient(140deg, var(--hbx-accent), var(--hbx-primary))"
      : tone === "success"
      ? "linear-gradient(140deg, var(--hbx-success), var(--hbx-secondary))"
      : "linear-gradient(140deg, var(--hbx-brand), var(--hbx-brand-strong))";

  return (
    <span
      style={{
        display: "inline-grid",
        placeItems: "center",
        width: size,
        height: size,
        flex: "0 0 auto",
        borderRadius: size <= 28 ? "var(--radius-xs)" : "var(--radius-sm)",
        overflow: "hidden",
        color: "#FFFFFF",
        fontFamily: "var(--font-display)",
        fontSize: Math.round(size * 0.36),
        fontWeight: 800,
        letterSpacing: "0.02em",
        background: src ? "var(--hbx-surface-raised)" : gradient,
        boxShadow: "var(--shadow-inset), 0 10px 22px -16px color-mix(in srgb, var(--hbx-brand) 60%, transparent)",
        ...style,
      }}
      {...rest}
    >
      {src ? (
        <img src={src} alt={name} style={{ width: "100%", height: "100%", objectFit: "cover" }} />
      ) : (
        initials || "?"
      )}
    </span>
  );
}
