import React from "react";

/**
 * StatusDot — small pulsing presence indicator.
 * tone: online (success) · busy (warning) · offline (muted) · live (brand).
 */
function StatusDot({ tone = "online", pulse = true, size = 10, label, style, ...rest }) {
  const color = COLORS[tone] || COLORS.online;
  const dot = (
    <span
      aria-label={label}
      style={{
        position: "relative",
        display: "inline-block",
        width: size,
        height: size,
        borderRadius: "999px",
        background: color,
        boxShadow: `0 0 0 4px color-mix(in srgb, ${color} 18%, transparent)`,
        ...style,
      }}
      {...rest}
    >
      {pulse ? (
        <span
          style={{
            position: "absolute",
            inset: 0,
            borderRadius: "999px",
            background: color,
            animation: "hbxPing 1.8s var(--ease-out-quint) infinite",
          }}
        />
      ) : null}
      <style>{"@keyframes hbxPing{0%{transform:scale(1);opacity:.7}70%,100%{transform:scale(2.6);opacity:0}}"}</style>
    </span>
  );

  if (!label) return dot;
  return (
    <span style={{ display: "inline-flex", alignItems: "center", gap: "0.45rem", fontFamily: "var(--font-body)", fontSize: "0.72rem", fontWeight: 700, color: "var(--text-muted)" }}>
      {dot}
      {label}
    </span>
  );
}

const COLORS = {
  online: "var(--hbx-success)",
  live: "var(--hbx-brand-strong)",
  busy: "var(--hbx-warning)",
  offline: "var(--hbx-muted)",
};
