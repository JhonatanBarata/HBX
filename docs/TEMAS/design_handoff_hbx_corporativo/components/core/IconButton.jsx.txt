import React from "react";

/**
 * IconButton — square glass control holding a single icon/glyph.
 * Same liquid-glass language as Button, sized for toolbars and rails.
 */
function IconButton({ children, label, size = 38, variant = "secondary", disabled = false, onClick, style, ...rest }) {
  const tint =
    variant === "primary"
      ? {
          color: "#FDFDFF",
          borderColor: "color-mix(in srgb, var(--hbx-primary) 52%, var(--hbx-accent))",
          background:
            "radial-gradient(circle at 22% 18%, rgba(255,255,255,0.28), transparent 24%)," +
            "linear-gradient(140deg, color-mix(in srgb, var(--hbx-primary) 90%, transparent), color-mix(in srgb, var(--hbx-accent) 64%, transparent))",
        }
      : variant === "ghost"
      ? {
          color: "var(--text-body)",
          borderColor: "transparent",
          background: "transparent",
        }
      : {
          color: "var(--text-strong)",
          borderColor: "var(--border-hairline)",
          background:
            "radial-gradient(circle at 18% 18%, rgba(255,255,255,0.3), transparent 24%)," +
            "linear-gradient(160deg, color-mix(in srgb, var(--hbx-surface) 90%, transparent), color-mix(in srgb, var(--hbx-surface-soft) 70%, transparent))",
        };

  return (
    <button
      type="button"
      aria-label={label}
      title={label}
      disabled={disabled}
      onClick={onClick}
      style={{
        display: "inline-grid",
        placeItems: "center",
        width: size,
        height: size,
        flex: "0 0 auto",
        border: "1px solid",
        borderRadius: "var(--control-radius)",
        cursor: disabled ? "not-allowed" : "pointer",
        opacity: disabled ? 0.5 : 1,
        boxShadow: variant === "ghost" ? "none" : "var(--shadow-inset), var(--shadow-xs)",
        backdropFilter: "blur(var(--blur-glass)) saturate(160%)",
        WebkitBackdropFilter: "blur(var(--blur-glass)) saturate(160%)",
        transition: "transform var(--motion-fast), box-shadow var(--motion-fast), background var(--motion-fast)",
        ...tint,
        ...style,
      }}
      {...rest}
    >
      {children}
    </button>
  );
}
