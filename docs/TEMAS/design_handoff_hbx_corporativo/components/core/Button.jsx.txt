 * tinted gradient by intent, and a soft far-throw shadow.
 */
function Button({

  variant = "primary",
  size = "md",
  disabled = false,
  iconLeft = null,
  iconRight = null,
  type = "button",
  onClick,
  style,
  ...rest
}) {
  const base = {
    position: "relative",
    isolation: "isolate",
    overflow: "hidden",
    display: "inline-flex",
    alignItems: "center",
    justifyContent: "center",
    gap: "0.42rem",
    minHeight: size === "sm" ? "32px" : "38px",
    padding: size === "sm" ? "0.42rem 0.68rem" : "0.52rem 0.82rem",
    fontFamily: "var(--font-body)",
    fontSize: size === "sm" ? "0.7rem" : "0.78rem",
    fontWeight: 800,
    lineHeight: 1,
    cursor: disabled ? "not-allowed" : "pointer",
    opacity: disabled ? 0.5 : 1,
    borderRadius: "calc(var(--control-radius) + 1px)",
    border: "1px solid",
    color: "var(--text-strong)",
    backdropFilter: "blur(calc(var(--blur-chrome) + 4px)) saturate(180%)",
    WebkitBackdropFilter: "blur(calc(var(--blur-chrome) + 4px)) saturate(180%)",
    transition: "transform var(--motion-fast), box-shadow var(--motion-fast), border-color var(--motion-fast), background var(--motion-fast)",
    ...VARIANTS[variant],
    ...style,
  };

  return (
    <button type={type} disabled={disabled} onClick={onClick} style={base} {...rest}>
      {iconLeft}
      {children}
      {iconRight}
    </button>
  );
}
