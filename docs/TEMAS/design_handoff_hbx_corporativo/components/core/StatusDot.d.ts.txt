import * as React from "react";

/** Pulsing presence indicator, optionally with a trailing label. */
export interface StatusDotProps extends React.HTMLAttributes<HTMLSpanElement> {
  /** @default "online" */
  tone?: "online" | "live" | "busy" | "offline";
  /** Animate the ping ring. @default true */
  pulse?: boolean;
  /** Dot diameter in px. @default 10 */
  size?: number;
  /** Optional text shown after the dot. */
  label?: string;
}

export function StatusDot(props: StatusDotProps): React.JSX.Element;
