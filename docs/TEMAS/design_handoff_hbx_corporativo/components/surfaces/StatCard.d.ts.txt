import * as React from "react";

/** Single-KPI tile with a monospaced value and optional delta. */
export interface StatCardProps extends React.HTMLAttributes<HTMLDivElement> {
  /** Metric name. */
  label: React.ReactNode;
  /** The number/value (rendered in IBM Plex Mono, tabular). */
  value: React.ReactNode;
  /** Optional change line, e.g. "+12% esta semana". */
  delta?: React.ReactNode;
  /** Color of the delta. @default "success" */
  deltaTone?: "success" | "danger" | "muted";
  /** Optional icon pinned top-right. */
  icon?: React.ReactNode;
}

export function StatCard(props: StatCardProps): React.JSX.Element;
