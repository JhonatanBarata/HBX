import * as React from "react";

/**
 * Liquid-glass panel with an optional eyebrow + title header and a corner spotlight.
 *
 * @startingPoint section="Surfaces" subtitle="Glass panel with header slot" viewport="700x300"
 */
export interface CardProps extends React.HTMLAttributes<HTMLElement> {
  /** Small uppercase label above the title. */
  eyebrow?: React.ReactNode;
  /** Heading text. */
  title?: React.ReactNode;
  /** Node pinned to the header's right (e.g. an IconButton). */
  action?: React.ReactNode;
  /** Corner spotlight tint. @default "default" */
  tone?: "default" | "brand" | "accent" | "warning";
  /** Inner padding in px. @default 18 */
  padding?: number;
  children?: React.ReactNode;
}

export function Card(props: CardProps): React.JSX.Element;
