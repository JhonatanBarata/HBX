import React from "react";

/**
 * Card — the HBX liquid-glass panel. Optional eyebrow + title header.
 * `tone` adds a tinted spotlight (brand / accent / warning) in the corner.
 */
function Card({ children, eyebrow, title, action, tone = "default", padding = 18, style, ...rest }) {
  const spotlight = SPOTLIGHTS[tone] || SPOTLIGHTS.default;
  return (
    <section
      style={{
        border: "1px solid var(--border-hairline)",
        borderRadius: "var(--panel-radius)",
        padding,
        background: spotlight,
        boxShadow: "var(--shadow-inset), var(--shadow-sm)",
        fontFamily: "var(--font-body)",
        color: "var(--text-body)",
        ...style,
      }}
      {...rest}
    >
      {(eyebrow || title || action) && (
        <header style={{ display: "flex", alignItems: "flex-start", justifyContent: "space-between", gap: "0.75rem", marginBottom: 12 }}>
          <div style={{ display: "grid", gap: 4 }}>
            {eyebrow && (
              <span style={{ fontFamily: "var(--font-display)", fontSize: "0.62rem", fontWeight: 800, letterSpacing: "0.16em", textTransform: "uppercase", color: "var(--text-muted)" }}>
                {eyebrow}
              </span>
            )}
            {title && (
              <h3 style={{ margin: 0, fontFamily: "var(--font-display)", fontSize: "1.02rem", fontWeight: 800, letterSpacing: "-0.03em", lineHeight: 1.1, color: "var(--text-strong)" }}>
                {title}
              </h3>
            )}
          </div>
          {action}
        </header>
      )}
      {children}
    </section>
  );
}

const SPOTLIGHTS = {
  default:
    "radial-gradient(circle at top right, color-mix(in srgb, var(--hbx-brand) 8%, transparent), transparent 48%)," +
    "linear-gradient(180deg, color-mix(in srgb, var(--hbx-surface) 98%, var(--hbx-background)), color-mix(in srgb, var(--hbx-surface-soft) 92%, var(--hbx-background)))",
  brand:
    "radial-gradient(circle at top right, color-mix(in srgb, var(--hbx-brand) 18%, transparent), transparent 42%)," +
    "linear-gradient(180deg, color-mix(in srgb, var(--hbx-surface) 96%, var(--hbx-background)), color-mix(in srgb, var(--hbx-surface-soft) 90%, var(--hbx-background)))",
  accent:
    "radial-gradient(circle at top right, color-mix(in srgb, var(--hbx-accent) 16%, transparent), transparent 42%)," +
    "linear-gradient(180deg, color-mix(in srgb, var(--hbx-surface) 96%, var(--hbx-background)), color-mix(in srgb, var(--hbx-surface-soft) 90%, var(--hbx-background)))",
  warning:
    "radial-gradient(circle at top right, color-mix(in srgb, var(--hbx-warning) 18%, transparent), transparent 42%)," +
    "linear-gradient(180deg, color-mix(in srgb, var(--hbx-surface) 96%, var(--hbx-background)), color-mix(in srgb, var(--hbx-surface-soft) 90%, var(--hbx-background)))",
};
