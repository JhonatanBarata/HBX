import React from "react";

/**
 * ChatBubble — a WhatsApp-style message bubble for the HBX atendimento inbox.
 * direction: "in" (customer), "out" (agent), "system" (bot / status note).
 */
function ChatBubble({ children, direction = "in", time, author, style, ...rest }) {
  if (direction === "system") {
    return (
      <div style={{ display: "flex", justifyContent: "center", margin: "0.4rem 0" }} {...rest}>
        <span style={{ padding: "0.3rem 0.7rem", borderRadius: "var(--radius-pill)", background: "var(--hbx-chat-system)", border: "1px solid var(--border-hairline)", fontFamily: "var(--font-body)", fontSize: "0.7rem", fontWeight: 600, color: "var(--text-muted)" }}>
          {children}
        </span>
      </div>
    );
  }

  const out = direction === "out";
  return (
    <div style={{ display: "flex", justifyContent: out ? "flex-end" : "flex-start", margin: "0.2rem 0" }} {...rest}>
      <div
        style={{
          maxWidth: "76%",
          padding: "0.55rem 0.75rem 0.4rem",
          borderRadius: out ? "16px 16px 4px 16px" : "16px 16px 16px 4px",
          background: out ? "var(--hbx-chat-outbound)" : "var(--hbx-chat-inbound)",
          border: "1px solid var(--border-hairline)",
          boxShadow: "var(--shadow-inset), var(--shadow-xs)",
          fontFamily: "var(--font-body)",
          ...style,
        }}
      >
        {author && !out && (
          <div style={{ fontSize: "0.68rem", fontWeight: 800, color: "var(--hbx-brand-strong)", marginBottom: 2 }}>{author}</div>
        )}
        <div style={{ fontSize: "0.84rem", lineHeight: 1.45, color: "var(--text-strong)" }}>{children}</div>
        {time && (
          <div className="hbx-mono" style={{ fontFamily: "var(--font-mono)", fontSize: "0.62rem", color: "var(--text-muted)", textAlign: "right", marginTop: 2 }}>{time}</div>
        )}
      </div>
    </div>
  );
}
