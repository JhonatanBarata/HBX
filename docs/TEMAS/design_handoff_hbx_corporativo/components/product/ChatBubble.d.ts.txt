import * as React from "react";

/** WhatsApp-style message bubble for the HBX atendimento inbox. */
export interface ChatBubbleProps extends React.HTMLAttributes<HTMLDivElement> {
  /** Who sent it. @default "in" */
  direction?: "in" | "out" | "system";
  /** Timestamp (rendered in mono). */
  time?: React.ReactNode;
  /** Sender name, shown only on inbound bubbles. */
  author?: React.ReactNode;
  children?: React.ReactNode;
}

export function ChatBubble(props: ChatBubbleProps): React.JSX.Element;
